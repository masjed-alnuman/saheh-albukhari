الحفظ التلقائي في Google Sheets & Google Drive

طريقة احترافية، مجانية 100%، ومحفوظة دائماً في حسابك الشخصي على غوغل درايف

كيف تربط الموقع بحسابك في Google Drive / Sheets في دقيقتين (مجاناً للأبد)؟

  1. افتح حسابك في Google Drive وأنشئ ملف Google Sheets جديد باسم: حضور مجلس صحيح البخاري - مسجد النعمان.
  2. من القائمة العلوية في الملف اختر: Extensions (الإضافات) ثم Apps Script.
  3. امسح أي كود موجود هناك، والصق الكود الجاهز بالأسفل ثم اضغط حفظ 💾.
  4. اضغط على زر Deploy (نشر) أعلى اليمين ← New deployment ← اختر النوع Web app، واجعل Who has access: Anyone (أي شخص)، ثم انسخ الرابط والصقه في الخانة أعلاه!
كود Google Apps Script الجاهز (يدعم العربية والألوان والتنسيق التلقائي) 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("سجل الحضور التراكمي") || ss.insertSheet("سجل الحضور التراكمي");
    sheet.clear();
    sheet.setRightToLeft(true);
    
    // Build Header Row
    var headers = ["الرقم التسلسلي", "الاسم", "اسم الأب", "الكنية", "الاسم الكامل", "إجمالي الحضور", "إجمالي الغياب"];
    data.weeks.forEach(function(w) { headers.push(w.label + " (" + w.date + ")"); });
    
    var rows = [headers];
    data.students.forEach(function(s) {
      var presentCount = 0, absentCount = 0;
      var weekCells = data.weeks.map(function(w) {
        var st = (s.attendance && s.attendance[w.id]) || "pending";
        if (st === "present") { presentCount++; return "✓ حاضر"; }
        if (st === "absent") { absentCount++; return "✗ غائب"; }
        return "— لم يسجل";
      });
      rows.push([s.serialNumber, s.firstName, s.fatherName, s.lastName, s.fullName, presentCount, absentCount].concat(weekCells));
    });
    
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    sheet.setFrozenRows(1);
    sheet.setFrozenColumns(5);
    sheet.getRange(1, 1, 1, headers.length).setBackground("#064e3b").setFontColor("#ffffff").setFontWeight("bold");
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
